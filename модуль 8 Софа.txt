from PyQt6.QtCore import Qt
from PyQt6.QtWidgets import *
import sys
import pymysql
from PyQt6.QtGui import QColor

class MainWin(QWidget):
    def __init__(self):
        super().__init__()

        self.con = pymysql.connect(host='localhost', user='root', password='root', database='orders_db')
        self.setWindowTitle('Статистика заказчиков')
        self.resize(1000,1000)
        self.layout = QGridLayout(self)
# Фильтр: лейбл, комбо, кнопка
        self.cust_label = QLabel('Выберите заказчика')
        self.layout.addWidget(self.cust_label, 1, 0)
        self.cust_combo = QComboBox()
        self.cust_combo.addItem('Показать всех', userData = None)
        self.cust_combo.setCurrentIndex(self.cust_combo.count() - 1)
        self.layout.addWidget(self.cust_combo, 1, 1)
        self.filter_btn = QPushButton('Фильтровать')
        self.filter_btn.clicked.connect(self.apply_filter)
        self.layout.addWidget(self.filter_btn, 1, 2)
# Поиск: лейбл, лайн, кнопка
        self.search_label = QLabel('Введите заказчика')
        self.layout.addWidget(self.search_label, 2, 0)
        self.search_line = QLineEdit()
        self.layout.addWidget(self.search_line, 2, 1)
        self.search_btn = QPushButton('Найти')
        self.search_btn.clicked.connect(self.search)
        self.layout.addWidget(self.search_btn, 2, 2)
# Сортировка: список, радио
        self.choose_label = QLabel('Выберите параметры сортировки')
        self.choose_label.setFixedHeight(30)
        self.layout.addWidget(self.choose_label, 0, 3, 1, 2)
        self.sort_list = QListWidget()
        self.sort_list.addItems(['Заказчик', 'Город','Дата заказа','Сумма заказа'])
        self.layout.addWidget(self.sort_list, 1, 3, 2, 2)
        self.sort_radio_asc = QRadioButton('По возрастанию')
        self.sort_radio_asc.setChecked(True)
        self.layout.addWidget(self.sort_radio_asc, 1, 5)
        self.sort_radio_desc = QRadioButton('По убыванию')
        self.layout.addWidget(self.sort_radio_desc, 2, 5)
# Таблица
        self.table = QTableWidget()
        self.table.setColumnCount(5)
        self.table.setHorizontalHeaderLabels(['Заказчик','Город','Телефон','Дата заказа','Сумма заказа'])
        self.table.horizontalHeader().setSectionResizeMode(QHeaderView.ResizeMode.Stretch)
        self.table.setEditTriggers(QTableWidget.EditTrigger.NoEditTriggers)
        self.layout.addWidget(self.table, 3, 0, 5, 6)
# Счетчики кол-ва заказов и суммы
        self.amount_qty_label = QLabel('Всего заказов: 0')
        self.amount_qty_label.setFixedHeight(30)
        self.layout.addWidget(self.amount_qty_label, 8, 0, 1, 2)
        self.amount_label = QLabel('Общая сумма заказов: 0.00')
        self.amount_label.setFixedHeight(30)
        self.layout.addWidget(self.amount_label, 9, 0, 1, 2)
# Загрузка комбо, таблицы, подключение сортировки
        self.load_customers()
        self.load_table_data()
        self.sort_list.currentRowChanged.connect(self.apply_sort)
        self.sort_radio_asc.toggled.connect(self.apply_sort)
        self.sort_radio_desc.toggled.connect(self.apply_sort)

    def load_customers(self):
        with self.con.cursor() as cur:
            cur.execute('SELECT DISTINCT customer FROM orders ORDER BY customer')
            for cust in cur.fetchall():
                self.cust_combo.addItem(cust[0], userData=cust[0])

    def load_table_data(self):
        with self.con.cursor() as cur:
            cur.execute("""
                    SELECT 
                        customer,
                        city,
                        phone,
                        DATE_FORMAT(order_date, '%d.%m.%Y') as order_date,
                        amount
                    FROM orders
                """)
            self.fill_table(cur.fetchall())

    def fill_table(self, data):
        self.table.clear()
        row_count = len(data)
        total_count = sum(row[4] for row in data)
        self.table.setRowCount(row_count)
        for row_num, row_data in enumerate(data):
            for col_num, cell_data in enumerate(row_data):
                self.table.setItem(row_num, col_num, QTableWidgetItem(str(cell_data)))
        self.amount_qty_label.setText(f'Всего заказов: {row_count}')
        self.amount_label.setText(f'Общая сумма заказов: {total_count}')

    def apply_filter(self):
        customer_data = self.cust_combo.currentData()
        if customer_data is None:
            self.load_table_data()
        else:
            with self.con.cursor() as cur:
                cur.execute('''SELECT customer, city, phone,
                DATE_FORMAT(order_date, '%%d.%%m.%%Y') as order_date,
                amount
                FROM orders
                WHERE customer = %s''',
                            (customer_data,))
                self.fill_table(cur.fetchall())

    def search(self):
        search_text = self.search_line.text().strip().lower()
        for row in range(self.table.rowCount()):
            for col in range(self.table.columnCount()):
                item = self.table.item(row, col)
                if item:
                    if search_text and search_text in item.text().lower():
                        item.setBackground(QColor('yellow'))
                    else:
                        item.setBackground(QColor('white'))

    def apply_sort(self):
        row = self.sort_list.currentRow()
        if row == -1 or self.table.rowCount() == 0:
            return
        column = [0, 1, 3, 4][row]
        if self.sort_radio_asc.isChecked():
            self.table.sortItems(column, Qt.SortOrder.AscendingOrder)
        else:
            self.table.sortItems(column, Qt.SortOrder.DescendingOrder)


if __name__ == '__main__':
    app = QApplication(sys.argv)
    win = MainWin()
    win.show()
    sys.exit(app.exec())
