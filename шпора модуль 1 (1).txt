DELIMITER $$
CREATE PROCEDURE create_dbs()
begin
declare i int default 1;
declare db_name VARCHAR(50);
declare user_name VARCHAR(50);
declare passwd VARCHAR(255);

while i <= 10 do
	SET db_name = CONCAT('BD',i);
    set user_name = concat('user',i);
    set passwd = SUBSTRING(MD5(RAND()), 1, 5);
    
    SET @sql = CONCAT('CREATE DATABASE IF NOT EXISTS`', db_name, '`');
    PREPARE s from @sql;
    execute s;
    deallocate prepare s;
    
    set @sql = concat('create user if not exists''', user_name, '''@''localhost'' identified by ''', passwd, '''');
    prepare s from @sql;
    execute s;
    deallocate prepare s;
    
    set @sql = concat('grant all privileges on `', db_name, '`.* to ''', user_name, '''@''localhost''');
    prepare s from @sql;
    execute s;
    deallocate prepare s;
    
    insert into Users(username, password_enc) values (user_name, passwd);
    set i = i + 1;
    
END WHILE;
end $$
DELIMITER ;

call create_dbs;
