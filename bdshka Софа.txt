-- =====================================================

-- Удаляем базу, если существует, и создаем заново
DROP DATABASE IF EXISTS MilkProduction;
CREATE DATABASE MilkProduction;
USE MilkProduction;

-- =====================================================
-- 1. ТАБЛИЦА Customers (Заказчики)
-- =====================================================
CREATE TABLE Customers (
    customer_id INT AUTO_INCREMENT PRIMARY KEY,
    external_id VARCHAR(20) UNIQUE,
    name VARCHAR(255) NOT NULL,
    inn VARCHAR(20),
    address TEXT,
    phone VARCHAR(20),
    is_salesman BOOLEAN DEFAULT FALSE,
    is_buyer BOOLEAN DEFAULT FALSE
);

-- =====================================================
-- 2. ТАБЛИЦА Materials (Материалы)
-- =====================================================
CREATE TABLE Materials (
    material_id INT AUTO_INCREMENT PRIMARY KEY,
    code VARCHAR(50) UNIQUE NOT NULL,
    name VARCHAR(255) NOT NULL,
    unit VARCHAR(20) NOT NULL,
    price DECIMAL(15, 2)
);

-- =====================================================
-- 3. ТАБЛИЦА Products (Продукция)
-- =====================================================
CREATE TABLE Products (
    product_id INT AUTO_INCREMENT PRIMARY KEY,
    code VARCHAR(50) UNIQUE NOT NULL,
    name VARCHAR(255) NOT NULL,
    unit VARCHAR(20) NOT NULL,
    price DECIMAL(15, 2)
);

-- =====================================================
-- 4. ТАБЛИЦА Specifications (Спецификации)
-- =====================================================
CREATE TABLE Specifications (
    spec_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    product_id INT NOT NULL,
    quantity_per_product DECIMAL(15, 3) DEFAULT 1,
    FOREIGN KEY (product_id) REFERENCES Products(product_id)
);

-- =====================================================
-- 5. ТАБЛИЦА SpecificationMaterials (Состав спецификации)
-- =====================================================
CREATE TABLE SpecificationMaterials (
    spec_material_id INT AUTO_INCREMENT PRIMARY KEY,
    spec_id INT NOT NULL,
    material_id INT NOT NULL,
    quantity DECIMAL(15, 5) NOT NULL,
    FOREIGN KEY (spec_id) REFERENCES Specifications(spec_id),
    FOREIGN KEY (material_id) REFERENCES Materials(material_id),
    UNIQUE KEY unique_spec_material (spec_id, material_id)
);

-- =====================================================
-- 6. ТАБЛИЦА Orders (Заказы покупателей)
-- =====================================================
CREATE TABLE Orders (
    order_id INT AUTO_INCREMENT PRIMARY KEY,
    order_number VARCHAR(50) UNIQUE NOT NULL,
    order_date DATE NOT NULL,
    customer_id INT NOT NULL,
    total_amount DECIMAL(15, 2) DEFAULT 0,
    FOREIGN KEY (customer_id) REFERENCES Customers(customer_id)
);

-- =====================================================
-- 7. ТАБЛИЦА OrderItems (Позиции заказа)
-- =====================================================
CREATE TABLE OrderItems (
    order_item_id INT AUTO_INCREMENT PRIMARY KEY,
    order_id INT NOT NULL,
    product_id INT NOT NULL,
    quantity DECIMAL(15, 3) NOT NULL,
    unit_price DECIMAL(15, 2) NOT NULL,
    FOREIGN KEY (order_id) REFERENCES Orders(order_id) ON DELETE CASCADE,
    FOREIGN KEY (product_id) REFERENCES Products(product_id)
);

-- =====================================================
-- 8. ТАБЛИЦА Production (Производственные партии)
-- =====================================================
CREATE TABLE Production (
    production_id INT AUTO_INCREMENT PRIMARY KEY,
    production_number VARCHAR(50) UNIQUE NOT NULL,
    production_date DATE NOT NULL
);

-- =====================================================
-- 9. ТАБЛИЦА ProductionProducts (Выпуск продукции)
-- =====================================================
CREATE TABLE ProductionProducts (
    production_product_id INT AUTO_INCREMENT PRIMARY KEY,
    production_id INT NOT NULL,
    product_id INT NOT NULL,
    quantity DECIMAL(15, 3) NOT NULL,
    FOREIGN KEY (production_id) REFERENCES Production(production_id) ON DELETE CASCADE,
    FOREIGN KEY (product_id) REFERENCES Products(product_id)
);

-- =====================================================
-- 10. ТАБЛИЦА ProductionMaterials (Расход материалов)
-- =====================================================
CREATE TABLE ProductionMaterials (
    production_material_id INT AUTO_INCREMENT PRIMARY KEY,
    production_id INT NOT NULL,
    material_id INT NOT NULL,
    quantity DECIMAL(15, 5) NOT NULL,
    FOREIGN KEY (production_id) REFERENCES Production(production_id) ON DELETE CASCADE,
    FOREIGN KEY (material_id) REFERENCES Materials(material_id)
);

-- =====================================================
-- ТРИГГЕР: Автоматический расчет итоговой суммы заказа
-- Срабатывает после добавления, обновления, удаления позиций
-- =====================================================

DELIMITER //

CREATE TRIGGER update_order_total_after_insert
AFTER INSERT ON OrderItems
FOR EACH ROW
BEGIN
    UPDATE Orders o
    SET o.total_amount = (
        SELECT COALESCE(SUM(oi.quantity * oi.unit_price), 0)
        FROM OrderItems oi
        WHERE oi.order_id = NEW.order_id
    )
    WHERE o.order_id = NEW.order_id;
END //

CREATE TRIGGER update_order_total_after_update
AFTER UPDATE ON OrderItems
FOR EACH ROW
BEGIN
    UPDATE Orders o
    SET o.total_amount = (
        SELECT COALESCE(SUM(oi.quantity * oi.unit_price), 0)
        FROM OrderItems oi
        WHERE oi.order_id = NEW.order_id
    )
    WHERE o.order_id = NEW.order_id;
END //

CREATE TRIGGER update_order_total_after_delete
AFTER DELETE ON OrderItems
FOR EACH ROW
BEGIN
    UPDATE Orders o
    SET o.total_amount = (
        SELECT COALESCE(SUM(oi.quantity * oi.unit_price), 0)
        FROM OrderItems oi
        WHERE oi.order_id = OLD.order_id
    )
    WHERE o.order_id = OLD.order_id;
END //

DELIMITER ;

-- =====================================================
-- ХРАНИМАЯ ПРОЦЕДУРА: Статистика за период
-- =====================================================

DELIMITER //

CREATE PROCEDURE GetOrderStatistics(
    IN p_start_date DATE,
    IN p_end_date DATE
)
BEGIN
    SELECT 
        c.name AS Заказчик,
        COUNT(DISTINCT o.order_id) AS Общее_количество_заказов,
        SUM(oi.quantity) AS Общее_количество_продукции,
        SUM(oi.quantity * oi.unit_price) AS Общая_сумма_заказов
    FROM Orders o
    JOIN Customers c ON o.customer_id = c.customer_id
    JOIN OrderItems oi ON o.order_id = oi.order_id
    WHERE o.order_date BETWEEN p_start_date AND p_end_date
    GROUP BY c.customer_id, c.name
    ORDER BY Общая_сумма_заказов DESC;
END //

DELIMITER ;

SELECT * FROM Orders;

-- Расчет себестоимости продукции по спецификации (с учетом цен материалов)
SELECT 
    p.name AS Продукция,
    SUM(sm.quantity * m.price) AS Себестоимость_материалов
FROM Products p
JOIN Specifications s ON p.product_id = s.product_id
JOIN SpecificationMaterials sm ON s.spec_id = sm.spec_id
JOIN Materials m ON sm.material_id = m.material_id
GROUP BY p.product_id;
