import pymysql
from PyQt6.QtWidgets import QMessageBox


class DatabaseConnection:
    def __init__(self):
        self.connection = None

    def connect(self):
        try:
            self.connection = pymysql.connect(host='localhost', user='root', password='root', database='orders_db')
            return self.connection
        except Exception as e:
            QMessageBox.critical(self, 'Database error', 'Ошибка подключения бд')

    def close(self):
        if self.connection:
            self.connection.close()

    def get_cursor(self):
        return self.connection.cursor()

from PyQt6.QtWidgets import *
from PyQt6.QtCore import Qt
from PyQt6.QtGui import QColor
from database.db_connection import DatabaseConnection


class App(QWidget):
    """Главное окно приложения"""

    def __init__(self):
        super().__init__()
        self.db = DatabaseConnection()
        self.table_model = None

        self.setWindowTitle('ООО Ассоль - Статистика заказчиков')
        self.setMinimumSize(800, 600)
        self.resize(1000, 700)

        self.init_ui()
        self.init_database()

    def init_ui(self):
        """Инициализация интерфейса"""
        layout = QGridLayout(self)
        layout.setSpacing(10)

        # Ряд 0: Заголовок сортировки
        sort_label = QLabel('Выберите параметры сортировки')
        sort_label.setFixedHeight(30)
        layout.addWidget(sort_label, 0, 3, 1, 2)

        # Ряд 1: Фильтр
        self.cust_label = QLabel('Выберите заказчика')
        layout.addWidget(self.cust_label, 1, 0)

        self.cust_combo = QComboBox()
        self.cust_combo.addItem('Показать всех', userData=None)
        layout.addWidget(self.cust_combo, 1, 1)

        self.filter_btn = QPushButton('Фильтровать')
        self.filter_btn.clicked.connect(self.on_filter_click)
        layout.addWidget(self.filter_btn, 1, 2)

        # Ряд 1: Сортировка
        self.sort_list = QListWidget()
        self.sort_list.setMaximumHeight(80)
        self.sort_list.addItems(['Заказчик', 'Город', 'Дата заказа', 'Сумма заказа'])
        layout.addWidget(self.sort_list, 1, 3, 2, 2)

        self.sort_radio_asc = QRadioButton('По возрастанию')
        self.sort_radio_desc = QRadioButton('По убыванию')
        self.sort_radio_asc.setChecked(True)
        layout.addWidget(self.sort_radio_asc, 1, 5)
        layout.addWidget(self.sort_radio_desc, 2, 5)

        # Ряд 2: Поиск
        self.search_label = QLabel('Поиск:')
        layout.addWidget(self.search_label, 2, 0)

        self.search_line = QLineEdit()
        self.search_line.setPlaceholderText('Введите текст для поиска...')
        layout.addWidget(self.search_line, 2, 1)

        self.search_btn = QPushButton('Найти')
        self.search_btn.clicked.connect(self.on_search_click)
        layout.addWidget(self.search_btn, 2, 2)

        # Ряд 3-7: Таблица (растягивается)
        self.table = QTableWidget()
        self.table.setColumnCount(5)
        self.table.setHorizontalHeaderLabels(['Заказчик', 'Город', 'Телефон', 'Дата заказа', 'Сумма заказа'])
        self.table.horizontalHeader().setSectionResizeMode(QHeaderView.ResizeMode.Stretch)
        self.table.setEditTriggers(QTableWidget.EditTrigger.NoEditTriggers)
        layout.addWidget(self.table, 3, 0, 5, 6)

        # Ряд 8-9: Статистика
        self.amount_qty_label = QLabel('Всего заказов: 0')
        self.amount_qty_label.setFixedHeight(30)
        layout.addWidget(self.amount_qty_label, 8, 0, 1, 2)

        self.amount_label = QLabel('Общая сумма заказов: 0.00')
        self.amount_label.setFixedHeight(30)
        layout.addWidget(self.amount_label, 9, 0, 1, 2)

        self.setLayout(layout)

    def init_database(self):
        """Инициализация базы данных"""
        try:
            self.db.connect()
            self.load_customers()
            self.load_table_data()

            # Подключение сигналов сортировки
            self.sort_list.currentRowChanged.connect(self.on_sort_changed)
            self.sort_radio_asc.toggled.connect(self.on_sort_changed)
            self.sort_radio_desc.toggled.connect(self.on_sort_changed)

        except Exception as e:
            QMessageBox.critical(self, "Ошибка", str(e))

    def load_customers(self):
        """Загрузка списка заказчиков"""
        try:
            cursor = self.db.get_cursor()
            cursor.execute('SELECT DISTINCT customer FROM orders ORDER BY customer')
            for row in cursor.fetchall():
                self.cust_combo.addItem(row[0], userData=row[0])
            cursor.close()
        except Exception as e:
            QMessageBox.critical(self, "Ошибка", f"Не удалось загрузить список заказчиков: {e}")

    def load_table_data(self):
        """Загрузка данных в таблицу"""
        try:
            cursor = self.db.get_cursor()
            cursor.execute("""
                SELECT customer, city, phone,
                       DATE_FORMAT(order_date, '%d.%m.%Y') as order_date,
                       amount
                FROM orders
            """)
            data = cursor.fetchall()
            self.fill_table(data)
            cursor.close()
        except Exception as e:
            QMessageBox.critical(self, "Ошибка", f"Не удалось загрузить данные: {e}")

    def fill_table(self, data):
        """Заполнение таблицы данными"""
        self.table.setRowCount(len(data))
        total_sum = 0

        for row_num, row_data in enumerate(data):
            self.table.setItem(row_num, 0, QTableWidgetItem(str(row_data[0])))
            self.table.setItem(row_num, 1, QTableWidgetItem(str(row_data[1])))
            self.table.setItem(row_num, 2, QTableWidgetItem(str(row_data[2])))
            self.table.setItem(row_num, 3, QTableWidgetItem(str(row_data[3])))
            self.table.setItem(row_num, 4, QTableWidgetItem(str(row_data[4])))
            total_sum += float(row_data[4])

        self.amount_qty_label.setText(f'Всего заказов: {len(data)}')
        self.amount_label.setText(f'Общая сумма заказов: {total_sum:,.2f}'.replace(',', ' '))

    def on_filter_click(self):
        """Обработчик кнопки фильтра"""
        customer_data = self.cust_combo.currentData()

        try:
            cursor = self.db.get_cursor()

            if customer_data is None:
                cursor.execute("""
                    SELECT customer, city, phone,
                           DATE_FORMAT(order_date, '%%d.%%m.%%Y') as order_date,
                           amount
                    FROM orders
                """)
            else:
                cursor.execute("""
                    SELECT customer, city, phone,
                           DATE_FORMAT(order_date, '%%d.%%m.%%Y') as order_date,
                           amount
                    FROM orders
                    WHERE customer = %s
                """, (customer_data,))

            self.fill_table(cursor.fetchall())
            cursor.close()
            self.clear_highlights()

        except Exception as e:
            QMessageBox.critical(self, "Ошибка", f"Ошибка фильтрации: {e}")

    def on_search_click(self):
        """Обработчик кнопки поиска"""
        search_text = self.search_line.text().strip().lower()

        for row in range(self.table.rowCount()):
            for col in range(self.table.columnCount()):
                item = self.table.item(row, col)
                if item:
                    if search_text and search_text in item.text().lower():
                        item.setBackground(QColor('yellow'))
                    else:
                        item.setBackground(QColor('white'))

    def on_sort_changed(self):
        """Обработчик изменения сортировки"""
        row = self.sort_list.currentRow()
        if row == -1:
            return

        field_map = {0: 'customer', 1: 'city', 2: 'order_date', 3: 'amount'}
        field = field_map[row]
        order = 'ASC' if self.sort_radio_asc.isChecked() else 'DESC'

        try:
            cursor = self.db.get_cursor()
            customer_data = self.cust_combo.currentData()

            if customer_data is None:
                cursor.execute(f"""
                    SELECT customer, city, phone,
                           DATE_FORMAT(order_date, '%%d.%%m.%%Y') as order_date,
                           amount
                    FROM orders
                    ORDER BY {field} {order}
                """)
            else:
                cursor.execute(f"""
                    SELECT customer, city, phone,
                           DATE_FORMAT(order_date, '%%d.%%m.%%Y') as order_date,
                           amount
                    FROM orders
                    WHERE customer = %s
                    ORDER BY {field} {order}
                """, (customer_data,))

            self.fill_table(cursor.fetchall())
            cursor.close()

        except Exception as e:
            QMessageBox.critical(self, "Ошибка", f"Ошибка сортировки: {e}")

    def clear_highlights(self):
        """Очистка подсветки"""
        for row in range(self.table.rowCount()):
            for col in range(self.table.columnCount()):
                item = self.table.item(row, col)
                if item:
                    item.setBackground(QColor('white'))

    def closeEvent(self, event):
        """Закрытие соединения с БД"""
        self.db.close()
        event.accept()
